### Step 1: TypeScript 인터페이스 import

### Step 2: 보드 생성 함수 만들기

```typescript
import type { BoardComponent } from '@/types/component-contract'
import { uid } from 'quasar'

/**
 * 새로운 보드 컴포넌트 생성
 */
export function createBoard(data: {
  name: string
  preset?: 'single' | 'split-lr' | 'l-shape' | 'split-tb'
  devices?: string[]
}): BoardComponent {
  const now = new Date().toISOString()
  
  // 표준 계약 준수
  const board: BoardComponent = {
    // 필수 필드
    id: uid(),  // 고유 ID 생성
    type: 'board',
    version: '1.0',
    
    // 메타데이터 (표준)
    metadata: {
      createdAt: now,
      updatedAt: now,
      author: 'current-user-id',  // 실제로는 현재 사용자 ID
      tags: [],
      description: data.name
    },
    
    // 보드 자율적 구조
    preset: data.preset || 'single',
    panes: [],
    devices: data.devices || [],
    
    // 표준 메서드
    toJSON() {
      return {
        id: this.id,
        type: this.type,
        version: this.version,
        metadata: this.metadata,
        preset: this.preset,
        panes: this.panes,
        devices: this.devices,
        actions: this.actions || [],
        dataSource: this.dataSource || null
      }
    }
  }
  
  return board
}
```

### Step 3: 보드 업데이트 함수

```typescript
/**
 * 보드 업데이트 (표준 계약 준수)
 */
export function updateBoard(
  board: BoardComponent,
  updates: Partial<Pick<BoardComponent, 'preset' | 'panes' | 'devices'>>
): BoardComponent {
  return {
    ...board,
    ...updates,
    // metadata.updatedAt 자동 업데이트
    metadata: {
      ...board.metadata,
      updatedAt: new Date().toISOString()
    }
  }
}
```

### Step 4: 실제 사용 예시

```typescript
// 보드 생성
const newBoard = createBoard({
  name: '대시보드 1',
  preset: 'split-lr',
  devices: ['device-001', 'device-002']
})

// 보드 업데이트
const updatedBoard = updateBoard(newBoard, {
  preset: 'l-shape'
})

// JSON으로 변환
const boardJSON = updatedBoard.toJSON()
console.log(boardJSON)
```

```typescript
import type { PanelComponent } from '@/types/component-contract'
import { uid } from 'quasar'

/**
 * 새로운 패널 컴포넌트 생성

    panelType: data.panelType,

```

### Step 2: 패널 이동 (인터랙션 표준화)

```typescript
/**
 * 패널 이동 (actions 배열로 표준화)
 */
export function movePanel(
  panel: PanelComponent,
  newPosition: { x: number; y: number }
): PanelComponent {
  return {
    ...panel,
    grid: {
      ...panel.grid,
      x: newPosition.x,
      y: newPosition.y
    },
    // 인터랙션을 actions 배열로 기록
    actions: [
      ...(panel.actions || []),
      {
        id: uid(),
        type: 'control' as const,
        target: panel.id,
        params: {
          action: 'move',
          x: newPosition.x,
          y: newPosition.y
        },
        enabled: true,
        label: '패널 이동'
      }
    ],
    metadata: {
      ...panel.metadata,
      updatedAt: new Date().toISOString()
    }
  }
}
```

## 예제 4: 기존 데이터에 표준 계약 적용하기

### Step 1: 변환 함수 만들기

```typescript
import type { BoardComponent } from '@/types/component-contract'

/**
 * 기존 보드 데이터를 표준 계약 형식으로 변환
 */
export function migrateBoardToStandardContract(oldBoard: {
  id: string
  name: string
  preset?: string
  devices?: string[]
  // 기타 기존 필드들...
}): BoardComponent {
  const now = new Date().toISOString()
  
  return {
    // 기존 필드 유지
    id: oldBoard.id,
    type: 'board',
    version: '1.0',  // 새로 추가
    
    // 메타데이터 생성 (기존 데이터에서 추출하거나 기본값)
    metadata: {
      createdAt: now,  // 기존 데이터에 없으면 현재 시간
      updatedAt: now,
      author: 'unknown',  // 기존 데이터에서 추출 가능하면 추출
      tags: [],
      description: oldBoard.name
    },
    
    // 기존 구조 유지
    preset: (oldBoard.preset as any) || 'single',
    panes: [],  // 기존 데이터에서 변환
    devices: oldBoard.devices || [],
    
    // 표준 메서드 추가
    toJSON() {
      return {
        id: this.id,
        type: this.type,
        version: this.version,
        metadata: this.metadata,
        preset: this.preset,
        panes: this.panes,
        devices: this.devices,
        actions: this.actions || [],
        dataSource: this.dataSource || null
      }
    }
  }
}
```

### Step 2: 기존 Store에 적용

```typescript
// src/stores/boardMenuStore.ts
import { migrateBoardToStandardContract } from '@/utils/boardMigration'
import type { BoardComponent } from '@/types/component-contract'

export const useBoardMenuStore = defineStore('boardMenu', () => {
  const nodes = ref<BoardComponent[]>([])
  
  function addNode(nodeData: {
    name: string
    type: 'group' | 'board'
    // 기타 필드...
  }) {
    if (nodeData.type === 'board') {
      // 새로 만들 때는 표준 계약 적용
      const board = createBoard({
        name: nodeData.name,
        preset: nodeData.preset || 'single',
        devices: nodeData.devices || []
      })
      nodes.value.push(board)
      return board
    }
    // group은 기존 로직 유지
  }
  
  function loadFromLocalStorage() {
    const stored = localStorage.getItem('boardMenuDataNexa')
    if (stored) {
      const oldData = JSON.parse(stored)
      // 기존 데이터를 표준 계약으로 변환
      nodes.value = oldData.map((node: any) => {
        if (node.type === 'board') {
          return migrateBoardToStandardContract(node)
        }
        return node  // group은 그대로
      })
    }
  }
})
```

## 예제 5: Vue 컴포넌트에서 사용하기

### Step 1: Vue 컴포넌트 예시

```vue
<template>
  <div>
    <q-btn @click="createNewBoard">새 보드 만들기</q-btn>
    <div v-for="board in boards" :key="board.id">
      {{ board.metadata.description }}
      <small>{{ board.version }}</small>
    </div>
  </div>
</template>

<script setup lang="ts">
import { ref } from 'vue'
import type { BoardComponent } from '@/types/component-contract'
import { createBoard, updateBoard } from '@/utils/boardUtils'

const boards = ref<BoardComponent[]>([])

function createNewBoard() {
  const newBoard = createBoard({
    name: '새 대시보드',
    preset: 'split-lr'
  })
  boards.value.push(newBoard)
  
  // JSON으로 저장
  const json = newBoard.toJSON()
  console.log('보드 JSON:', json)
}

function updateBoardPreset(board: BoardComponent, newPreset: 'single' | 'split-lr') {
  const index = boards.value.findIndex(b => b.id === board.id)
  if (index !== -1) {
    boards.value[index] = updateBoard(board, { preset: newPreset })
  }
}
</script>
```

## 체크리스트: 새 컴포넌트 만들 때

새 컴포넌트를 만들 때 다음을 확인하세요:

- [ ] `@/types/component-contract`에서 인터페이스 import
- [ ] 필수 필드 포함: `id`, `type`, `version`
- [ ] `metadata` 객체 포함: `createdAt`, `updatedAt`
- [ ] `toJSON()` 메서드 구현
- [ ] 인터랙션이 있으면 `actions` 배열로 표준화
- [ ] 데이터 소스가 있으면 `dataSource` 객체로 표준화
- [ ] TypeScript 타입 적용 (선택적)
